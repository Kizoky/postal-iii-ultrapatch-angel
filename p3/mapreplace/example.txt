// This is an example of how mapreplace work
// the txt files must match with the map name, for example if you want to replace something from the main menu
// then the txt file must be main_menu.txt

"MapReplace"
{
	// this will replace main menu dude because targetname is "dud"
	"dud"
	{
		"StartDisabled" "0"
		"spawnflags" "0"
		"solid" "6"
		"skin" "0"
		"SetBodyGroup" "0"
		"rendermode" "0"
		"renderfx" "0"
		"rendercolor" "255 255 255"
		"renderamt" "255"
		"RandomAnimation" "0"
		"pressuredelay" "0"
		"PerformanceMode" "0"
		"model" "models/main_menu/dude_main_menu_v2.mdl"
		"mindxlevel" "0"
		"MinAnimTime" "5"
		"maxdxlevel" "0"
		"MaxAnimTime" "10"
		"fadescale" "1"
		"fademindist" "-1"
		"fademaxdist" "0"
		"ExplodeRadius" "0"
		"ExplodeDamage" "0"
		"disableshadows" "0"
		"disablereceiveshadows" "0"
		"angles" "0 270 0"
		"classname" "prop_dynamic"
	}
	
	// this will replace every box with matching model
	"models/containers/box_03.mdl"
	{
		"classname"	"prop_vehicle_airboat"
		"model"		"models/airboat.mdl"
	}

	// this will replace EVERY prop_dynamic that doesn't have a targetname
	"prop_dynamic"
	{
		"classname"	"prop_vehicle_airboat"
		"model"		"models/airboat.mdl"
	}
}
