// The type for the stats by default is integer
// You can specify 3 types:
// "Type"	"integer"
// "Type"	"float"
// "Type"	"boolean"

// Stats supports multiple languages, you can use 
// "Display"	"#STATS_KILLS"

"stats"
{
	// name of the stat
	"kills"
	{
		// The text to display
		"Display"	"#STATS_KILLS"
	}
	
	"human_kills"
	{
		"Display"	"#STATS_HUMAN_KILLS"
	}
	
	"animal_kills"
	{
		"Display"	"#STATS_ANIMAL_KILLS"
	}
	
	"headshots"
	{
		"Display"	"#STATS_HEADSHOTS"
	}
}
