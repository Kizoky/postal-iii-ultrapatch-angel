// This file is used to create inventory items, Angel by default has no inventory items! (Use this file as a guide)
// Make sure to have an unique namespace, since it will be used for stacking items!

"inventory_item"
{
	// Unique namespace, shouldn't conflict with other mods
	"angel"
	{
		// Name of the inventory item, this will be used to create inventory items on spawn or at a later time
		// This should be referred to angel::pizza in maps and AngelScript (both server and client)
		"pizza"
		{
			"behavior"		"bh_angelitem_pizza" 				// Postal3Script behavior
			"faction"		"Items"				 				// Postal3Script faction
			"manner"		"itmFood"			 				// Postal3Script manner
			"initState"		"st_init"			 				// The init state of bh_angelitem_pizza
			"startState"	"st_start"			 				// The start state of bh_angelitem_pizza
			"stacking"		"1"					 				// This item can be stacked, won't occupy multiple inventory slots when picked up multiple times
			"maxStacking"	"7"					 				// The maximum number of stacking, Player won't be able to pick up any more of this type of item
																// and won't occupy another slot
												 
			"canBeDropped"	"1"					 				// Whether the item could be dropped by the Player or not
			"model"			"models/small_things/pizza_01.mdl"	// The inventory item's new model, this can be left blank to use the model provided by the mapper or scripter
			"icon"			"hud/items/pizza.vtf"				// Used by client, the icon for the inventory item to display
		}
		
		"myseconditem"
		{
			"behavior"		"bh_angelitem_pizza" 				// Postal3Script behavior
			"faction"		"Items"				 				// Postal3Script faction
			"manner"		"itmFood"			 				// Postal3Script manner
			"initState"		"st_init"			 				// The init state of bh_angelitem_pizza
			"startState"	"st_start"			 				// The start state of bh_angelitem_pizza
			"stacking"		"1"					 				// This item can be stacked, won't occupy multiple inventory slots when picked up multiple times
			"maxStacking"	"7"					 				// The maximum number of stacking, Player won't be able to pick up any more of this type of item
																// and won't occupy another slot
												 
			"canBeDropped"	"1"					 				// Whether the item could be dropped by the Player or not
			"model"			"models/small_things/pizza_01.mdl"	// The inventory item's new model, this can be left blank to use the model provided by the mapper or scripter
			"icon"			"hud/items/pizza.vtf"				// Used by client, the icon for the inventory item to display
		}
	}
}

// Inventory items must use 'wj_prop_fsmitem' prop in maps or scripts
// This prop has a keyfield called 'InventoryItem', for the pizza example above this should be 'angel::pizza'
// do note that you must always use a namespace!

// for scripters, create the entity as normal, but before spawning it call the function "TransformItem"
// inside AngelScript this would be for the example above:
// pointer.TransformItem("angel::pizza");
// 
// If you do not want to use this, you can modify the properties just before spawning it, check out our AngelScript Docs for this.
